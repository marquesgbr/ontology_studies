# Nota de execução: Consultas com cláusula de classe superior
# (a ont:KMeans, a ont:TreeBased, a ont:ArtificialNeuralNetwork,
#  a ont:DeepLearningClustering) assumem raciocinador ativo no Protégé
# (Snap SPARQL + HermiT) ou inferência RDFS/OWL; alternativamente, pode-se
# substituir "a" por "rdf:type/rdfs:subClassOf*" em motores SPARQL 1.1 sem
# raciocinador OWL.

# ------------------------------------------------------------------------------
# CQ 1 (Resgatada): Quais algoritmos (incluindo ArtificialNeuralNetwork e
# DeepLearningClustering) utilizam Regra da Cadeia e Descida do Gradiente?
# ------------------------------------------------------------------------------
PREFIX ont: <http://www.semanticweb.org/gabrielstamford/ontologies/2026/8/MLOntology#>

SELECT DISTINCT ?algoritmo
WHERE {
  ?algoritmo ont:usesMathematicalFoundation ont:ChainRule ;
             ont:usesMathematicalFoundation ont:GradientDescent .
  {
    ?algoritmo a ont:ArtificialNeuralNetwork .
  }
  UNION
  {
    ?algoritmo a ont:DeepLearningClustering .
  }
}
ORDER BY ?algoritmo


# ------------------------------------------------------------------------------
# CQ 2: Quais são os fundamentos matemáticos mais utilizados pelos modelos de
# Machine Learning na base consolidada?
# ------------------------------------------------------------------------------
PREFIX ont: <http://www.semanticweb.org/gabrielstamford/ontologies/2026/8/MLOntology#>

SELECT ?teoriaMatematica (COUNT(DISTINCT ?algoritmo) AS ?totalUsos)
WHERE {
  ?algoritmo ont:usesMathematicalFoundation ?teoriaMatematica .
}
GROUP BY ?teoriaMatematica
ORDER BY DESC(?totalUsos) ?teoriaMatematica


# ------------------------------------------------------------------------------
# CQ 3 (Resgatada): Qual é a distribuição e predominância dos paradigmas de
# aprendizado (MLTypes) entre os algoritmos modelados?
# ------------------------------------------------------------------------------
PREFIX ont: <http://www.semanticweb.org/gabrielstamford/ontologies/2026/8/MLOntology#>

SELECT ?paradigma (COUNT(DISTINCT ?algoritmo) AS ?totalAlgoritmos)
WHERE {
  ?algoritmo ont:utilizesMLType ?paradigma .
}
GROUP BY ?paradigma
ORDER BY DESC(?totalAlgoritmos) ?paradigma


# ------------------------------------------------------------------------------
# CQ 4: Quais aplicações são resolvidas por algoritmos da família K-Means?
# ------------------------------------------------------------------------------
PREFIX ont: <http://www.semanticweb.org/gabrielstamford/ontologies/2026/8/MLOntology#>

SELECT DISTINCT ?aplicacao
WHERE {
  ?aplicacao ont:isSolvedBy ?algoritmo .
  ?algoritmo a ont:KMeans .
}
ORDER BY ?aplicacao


# ------------------------------------------------------------------------------
# CQ 5 (Resgatada): Quais algoritmos utilizam conceitos de Entropia ou
# Regularização por Entropia?
# ------------------------------------------------------------------------------
PREFIX ont: <http://www.semanticweb.org/gabrielstamford/ontologies/2026/8/MLOntology#>

SELECT DISTINCT ?algoritmo ?conceitoEntropia
WHERE {
  ?algoritmo ont:usesMathematicalFoundation ?conceitoEntropia .
  FILTER (?conceitoEntropia = ont:Entropy || ?conceitoEntropia = ont:EntropyRegularization)
}
ORDER BY ?conceitoEntropia ?algoritmo


# ------------------------------------------------------------------------------
# CQ 6 (Data Properties Booleanas): Quais algoritmos de agrupamento não exigem
# definição prévia de k e lidam com ruído/outliers?
# ------------------------------------------------------------------------------
PREFIX ont: <http://www.semanticweb.org/gabrielstamford/ontologies/2026/8/MLOntology#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT ?algoritmo
WHERE {
  ?algoritmo ont:requiresClusterCountK "false"^^xsd:boolean ;
             ont:handlesNoiseOutliers  "true"^^xsd:boolean .
}
ORDER BY ?algoritmo


# ------------------------------------------------------------------------------
# CQ 7 (Data Property Enum/String): Quais métricas de avaliação devem ser
# maximizadas versus minimizadas?
# ------------------------------------------------------------------------------
PREFIX ont: <http://www.semanticweb.org/gabrielstamford/ontologies/2026/8/MLOntology#>

SELECT ?metrica ?direcao
WHERE {
  ?metrica ont:hasOptimalDirection ?direcao .
}
ORDER BY ?direcao ?metrica


# ------------------------------------------------------------------------------
# CQ 8 (TreeBased): Quais algoritmos baseados em árvores combinam Expansão de
# Taylor de segunda ordem e regularização L2?
# ------------------------------------------------------------------------------
PREFIX ont: <http://www.semanticweb.org/gabrielstamford/ontologies/2026/8/MLOntology#>

SELECT ?algoritmo
WHERE {
  ?algoritmo a ont:TreeBased ;
             ont:usesMathematicalFoundation ont:TaylorExpansion ;
             ont:usesMathematicalFoundation ont:L2_Reg .
}
ORDER BY ?algoritmo


# ------------------------------------------------------------------------------
# CQ 9 (TreeBased / Aplicações): Quais domínios de aplicação são resolvidos por
# modelos baseados em árvores e quais algoritmos os resolvem?
# ------------------------------------------------------------------------------
PREFIX ont: <http://www.semanticweb.org/gabrielstamford/ontologies/2026/8/MLOntology#>

SELECT ?aplicacao ?algoritmo
WHERE {
  ?aplicacao ont:isSolvedBy ?algoritmo .
  ?algoritmo a ont:TreeBased .
}
ORDER BY ?aplicacao ?algoritmo


# ------------------------------------------------------------------------------
# CQ 10 (Nova — ArtificialNeuralNetwork / Aplicações): Quais domínios de
# aplicação são resolvidos por modelos da classe ArtificialNeuralNetwork?
# ------------------------------------------------------------------------------
PREFIX ont: <http://www.semanticweb.org/gabrielstamford/ontologies/2026/8/MLOntology#>

SELECT ?aplicacao ?algoritmo
WHERE {
  ?aplicacao ont:isSolvedBy ?algoritmo .
  ?algoritmo a ont:ArtificialNeuralNetwork .
}
ORDER BY ?aplicacao ?algoritmo
